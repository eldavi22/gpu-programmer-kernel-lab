cmake_minimum_required(VERSION 3.24)
project(gpu_programmer_kernel_lab LANGUAGES CXX)
option(ENABLE_CUDA "Build the CUDA tiled GEMM backend" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
add_library(gpu_gemm_cpu src/cpu_gemm.cpp)
target_include_directories(gpu_gemm_cpu PUBLIC include)
target_compile_options(gpu_gemm_cpu PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gpu-gemm src/main.cpp)
target_link_libraries(gpu-gemm PRIVATE gpu_gemm_cpu)
target_include_directories(gpu-gemm PRIVATE include)
add_executable(test_cpu tests/test_cpu.cpp)
target_link_libraries(test_cpu PRIVATE gpu_gemm_cpu)
target_include_directories(test_cpu PRIVATE include)
enable_testing()
add_test(NAME cpu_reference COMMAND test_cpu)
if(ENABLE_CUDA)
  enable_language(CUDA)
  add_library(gpu_gemm_cuda src/cuda_gemm.cu)
  target_compile_definitions(gpu_gemm_cpu PUBLIC GPU_GEMM_ENABLE_CUDA)
  target_compile_definitions(gpu-gemm PRIVATE GPU_GEMM_ENABLE_CUDA)
  target_include_directories(gpu_gemm_cuda PUBLIC include)
  target_link_libraries(gpu-gemm PRIVATE gpu_gemm_cuda)
  set_target_properties(gpu_gemm_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
